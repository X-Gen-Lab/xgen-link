cmake_minimum_required(VERSION 3.24)

include(Dependencies.cmake)
xgl_dev_validate_sources()
project(xgl_development VERSION 3.0.0 LANGUAGES C)

set(XGL_PROFILE "full" CACHE STRING "Protocol profile: boot, embedded, full")
set_property(CACHE XGL_PROFILE PROPERTY STRINGS boot embedded full)
set(host_default ON)
set(heap_default OFF)
if(XGL_PROFILE STREQUAL "boot")
    set(host_default OFF)
elseif(XGL_PROFILE STREQUAL "full")
    set(heap_default ON)
elseif(NOT XGL_PROFILE STREQUAL "embedded")
    message(FATAL_ERROR "Unknown XGL_PROFILE '${XGL_PROFILE}'")
endif()

option(XGL_BUILD_TESTS "Build protocol GoogleTest regression tests" ${host_default})
option(XGL_BUILD_EXAMPLES "Build protocol examples" ${host_default})
option(XGL_BUILD_DOCS "Build the documentation site" OFF)
option(XGL_ALLOW_FALLBACK_MALLOC "Enable explicit libc allocator compatibility" ${heap_default})
option(XGL_BUILD_NOHEAP_SMOKE "Build bounded lifecycle checks" ON)
option(XGL_BUILD_FOOTPRINT_REPORT "Provide protocol archive size reporting" ON)
option(XGL_BUILD_STATIC_ANALYSIS_TARGET "Provide static analysis target" ON)
option(XGL_BUILD_SDK_CONSUMER_SMOKE "Test installed package consumption" ON)
option(XGL_BUILD_RELEASE_VALIDATION_TARGET "Provide aggregate release checks" ON)
if(XGL_PROFILE STREQUAL "boot" AND (XGL_BUILD_TESTS OR XGL_ALLOW_FALLBACK_MALLOC))
    message(FATAL_ERROR "Boot excludes GoogleTest capabilities and fallback malloc")
endif()

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
xgl_dev_provide_components()
set(XGL_SDK_INSTALL_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." link)
